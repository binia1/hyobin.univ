  const [registered, setRegistered] = useState([]);
  const [isLoading, setIsLoading] = useState(false);
  const [toasts, setToasts] = useState([]);

  // 실시간 서버 시간 업데이트
  useEffect(() => {
    const timer = setInterval(() => setServerTime(new Date()), 1000);
    return () => clearInterval(timer);
  }, []);

  const formatTime = (date) => {
    return date.toLocaleTimeString('ko-KR', { hour12: false });
  };

  const addToast = (msg, type = 'success') => {
    const id = Date.now();
    setToasts((prev) => [...prev, { id, msg, type }]);
    setTimeout(() => {
      setToasts((prev) => prev.filter((t) => t.id !== id));
    }, 3000);
  };

  const filterCourses = () => {
    return COURSES_DATA.filter((c) => {
      const matchDept = filterDept === 'ALL' || c.dept === filterDept;
      const matchType = filterType === 'ALL' || c.type === filterType;
      const matchSearch =
        c.title.toLowerCase().includes(searchKeyword.toLowerCase()) ||
        c.prof.includes(searchKeyword) ||
        c.id.toLowerCase().includes(searchKeyword.toLowerCase());
      return matchDept && matchType && matchSearch;
    });
  };

  const checkTimeOverlap = (newTime) => {
    const day = newTime[0];
    const periods = newTime.substring(1).split(',').map(Number);
    return registered.some((r) => {
      if (r.time[0] !== day) return false;
      const rPeriods = r.time.substring(1).split(',').map(Number);
      return periods.some((p) => rPeriods.includes(p));
    });
  };

  const handleApply = async (course) => {
    const totalCredits = registered.reduce((sum, r) => sum + r.credit, 0);
    
    if (registered.some((r) => r.id === course.id)) return;
    if (totalCredits + course.credit > 21) {
      addToast('최대 신청 학점(21.0)을 초과할 수 없습니다.', 'error');
      return;
    }
    if (checkTimeOverlap(course.time)) {
      addToast(`이미 같은 시간대에 다른 수업(${course.time})이 있습니다.`, 'error');
      return;
    }
    if (course.current >= course.max) {
      addToast('해당 강좌의 여석이 없습니다.', 'error');
      return;
    }

    setIsLoading(true);
    await new Promise((r) => setTimeout(r, 600)); // 모의 지연
    
    setRegistered((prev) => [...prev, course]);
    course.current += 1;
    setIsLoading(false);
    addToast(`[${course.title}] 신청 완료!`);
  };

  const handleCancel = (courseId) => {
    const course = COURSES_DATA.find((c) => c.id === courseId);
    setRegistered((prev) => prev.filter((r) => r.id !== courseId));
    if (course) course.current -= 1;
    addToast('수강 신청이 취소되었습니다.', 'info');
  };

  const totalCredits = registered.reduce((sum, r) => sum + r.credit, 0);

            </div>

        {/* 필터 및 검색 바 */}

            </select>
          </div>

            <span className="text-[11px] font-black text-slate-400 uppercase ml-1 tracking-wider">Search Keywords</span>
            <div className="relative">
              <input 
                type="text" 
                value={searchKeyword}
                onChange={(e) => setSearchKeyword(e.target.value)}

              />
              <Search className="absolute left-4 top-3 text-slate-400" size={18} />
            </div>
          </div>

          {/* 대시보드 스코어 */}
          <div className="flex items-center gap-6 bg-slate-900 text-white px-8 py-4 rounded-[28px] shadow-xl">
            <div className="text-center">
              <span className="text-[10px] text-slate-400 uppercase font-black block mb-0.5">Credit Limit</span>
              <span className="font-black text-xl leading-none tracking-tighter">21.0</span>
            </div>
            <div className="h-10 w-px bg-slate-700"></div>
            <div className="text-center">
              <span className="text-[10px] text-slate-400 uppercase font-black block mb-0.5">Registered</span>

          </div>

            <div className="overflow-y-auto flex-grow custom-scroll bg-white">

                <table className="w-full text-sm text-center border-collapse">
                  <thead className="sticky top-0 z-20 bg-slate-50 shadow-sm border-b">
                    <tr className="text-[10px] uppercase font-black text-slate-400">
                      <th className="p-4 w-24">Enrol</th>
                      <th className="p-4 w-24">Type</th>
                      <th className="p-4 text-left">Course Name</th>
                      <th className="p-4 w-16">CR</th>
                      <th className="p-4 w-24">Professor</th>
                      <th className="p-4 text-left">Time / Venue</th>

                    {filterCourses().map((course) => {

                          <td className="p-4">
                            <button 
                              onClick={() => isRegistered ? handleCancel(course.id) : handleApply(course)}
                              disabled={!isRegistered && isFull}

                                ? 'bg-red-50 text-red-500 border-2 border-red-200 hover:bg-red-500 hover:text-white' 
                                : isFull 
                                ? 'bg-slate-100 text-slate-300 cursor-not-allowed border-2 border-slate-100' 
                                : 'bg-white border-2 border-[#3344aa] text-[#3344aa] hover:bg-[#3344aa] hover:text-white hover:shadow-md active:scale-95'
                              }`}
                            >
                              {isRegistered ? 'CANCEL' : isFull ? 'CLOSED' : 'APPLY'}
                            </button>
                          </td>
                          <td className="p-4">
                            <span className={`px-2.5 py-1 rounded-lg text-[10px] font-black ${
                              course.type === '전필' ? 'bg-blue-100 text-blue-700' :
                              course.type === '교필' ? 'bg-amber-100 text-amber-700' :
                              'bg-slate-100 text-slate-600'
                            }`}>{course.type}</span>
                          </td>
                          <td className="p-4 text-left">
                            <div className="font-black text-slate-800 text-base">{course.title}</div>
                            <div className="text-[10px] text-slate-400 font-bold uppercase tracking-widest mt-0.5">{course.id}</div>
                          </td>
                          <td className="p-4 font-black text-[#3344aa] text-lg">{course.credit.toFixed(1)}</td>
                          <td className="p-4 font-bold text-slate-500">{course.prof}</td>
                          <td className="p-4 text-left">
                            <div className="text-[11px] font-black text-slate-700 flex items-center gap-1"><Clock size={10} /> {course.time}</div>
                            <div className="text-[10px] text-slate-400 font-bold uppercase mt-1 flex items-center gap-1"><MapPin size={10} /> {course.room}</div>
                          </td>
                          <td className="p-4">
                            <div className="flex flex-col items-center">
                              <div className="flex justify-between w-full max-w-[120px] mb-1.5">
                                <span className={`text-[10px] font-black ${isFull ? 'text-red-500' : 'text-[#3344aa]'}`}>{course.current} / {course.max}</span>
                                <span className="text-[10px] text-slate-400 font-bold">{Math.round((course.current/course.max)*100)}%</span>

                                <div className="h-full bg-[#3344aa] transition-all duration-700" style={{ width: `${(course.current/course.max)*100}%` }}></div>
                              </div>
                            </div>
                          </td>
                        </tr>
                      );
                    })}
                  </tbody>
                </table>
              </div>

              <div className="lg:hidden p-3 space-y-4 bg-slate-50/50">
                {filterCourses().map((course) => {
                  const isRegistered = registered.some((r) => r.id === course.id);
                  const isFull = course.current >= course.max;
                  return (
                    <div key={course.id} className={`bg-white p-5 rounded-[28px] border-2 shadow-sm animate-in transition-all ${isRegistered ? 'border-[#3344aa] ring-4 ring-blue-50' : 'border-slate-100'}`}>
                      <div className="flex justify-between items-start mb-4">
                        <div className="flex-grow pr-4">
                          <div className="flex items-center gap-2 mb-1.5">
                            <span className="text-[9px] font-black px-2 py-0.5 rounded-lg bg-[#3344aa]/10 text-[#3344aa] uppercase">{course.type}</span>
                            <span className="text-[9px] font-bold text-slate-300 font-mono tracking-tighter">{course.id}</span>
                          </div>
                          <h4 className="font-black text-slate-800 text-lg leading-tight">{course.title}</h4>
                        </div>
                        <button 
                          onClick={() => isRegistered ? handleCancel(course.id) : handleApply(course)}
                          disabled={!isRegistered && isFull}
                          className={`flex-shrink-0 w-20 py-3 rounded-2xl text-[11px] font-black shadow-lg transition-all ${
                            isRegistered ? 'bg-red-500 text-white shadow-red-200' : 
                            isFull ? 'bg-slate-100 text-slate-300 shadow-none' : 
                            'bg-[#3344aa] text-white shadow-blue-200 active:scale-90'
                          }`}
                        >
                          {isRegistered ? 'CANCEL' : isFull ? 'CLOSED' : 'APPLY'}
                        </button>
                      </div>
                      <div className="grid grid-cols-2 gap-y-3 mb-5 border-t border-slate-50 pt-4">
                        <div className="flex items-center gap-2 text-[12px] font-bold text-slate-500"><User size={14} className="text-[#3344aa]" /> {course.prof}</div>
                        <div className="flex items-center gap-2 text-[12px] font-bold text-slate-500"><Clock size={14} className="text-[#3344aa]" /> {course.time}</div>
                        <div className="flex items-center gap-2 text-[12px] font-bold text-slate-500"><MapPin size={14} className="text-[#3344aa]" /> {course.room}</div>
                        <div className="flex items-center gap-2 text-[12px] font-black text-[#3344aa]"><Award size={14} /> {course.credit.toFixed(1)} Credits</div>
                      </div>
                      <div className="bg-slate-50 p-4 rounded-2xl flex items-center justify-between">
                        <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">Enrolment Status</span>
                        <div className="flex items-center gap-4">
                           <span className={`text-[12px] font-black ${isFull ? 'text-red-500' : 'text-[#3344aa]'}`}>{course.current} / {course.max}</span>
                           <div className="w-24 h-2.5 bg-white rounded-full overflow-hidden border p-0.5">
                              <div className="h-full bg-[#3344aa] rounded-full" style={{ width: `${(course.current/course.max)*100}%` }}></div>
                           </div>
                        </div>
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          </div>

          {/* 우측 사이드바: 등록 목록 & 시간표 */}
          <div className="lg:w-[480px] flex-shrink-0 flex flex-col gap-4 overflow-hidden">
            {/* 장바구니 */}
            <div className="h-[40%] bg-white rounded-[32px] shadow-sm border border-slate-200 flex flex-col overflow-hidden">
              <div className="bg-slate-900 text-white px-6 py-4 flex justify-between items-center flex-shrink-0">
                <h3 className="font-black flex items-center text-xs uppercase tracking-widest">
                  <ShoppingCart className="text-yellow-400" size={16} /> Registered Subjects
                </h3>
                <span className="text-[11px] bg-[#3344aa] text-white px-3 py-1 rounded-xl font-black">
                  {registered.length} / 8
                </span>
              </div>
              
              <div className="overflow-y-auto flex-grow custom-scroll px-2 py-1">
                {registered.length === 0 ? (
                  <div className="h-full flex flex-col items-center justify-center text-slate-300 p-10 text-center">
                    <div className="w-20 h-20 bg-slate-50 rounded-[32px] flex items-center justify-center mb-4 border-2 border-dashed border-slate-200">
                      <PlusCircle className="opacity-20" size={32} />
                    </div>
                    <p className="text-[11px] font-black uppercase tracking-widest">Basket is empty</p>
                  </div>
                ) : (
                  <table className="w-full text-xs text-center border-collapse">
                    <thead className="bg-white text-slate-400 sticky top-0 border-b z-10">
                      <tr className="text-[9px] uppercase font-black">
                        <th className="p-3 w-10">Del</th>
                        <th className="p-3 text-left">Course Title</th>
                        <th className="p-3 w-12">CR</th>
                        <th className="p-3 w-24">Time</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-50">
                      {registered.map((r) => (
                        <tr key={r.id} className="animate-in group">
                          <td className="p-3">
                            <button onClick={() => handleCancel(r.id)} className="text-slate-300 hover:text-red-500 transition-all transform hover:scale-125">
                              <MinusCircle size={20} />
                            </button>
                          </td>
                          <td className="p-3 text-left font-black text-slate-700 truncate max-w-[150px]">{r.title}</td>
                          <td className="p-3 font-black text-[#3344aa]">{r.credit.toFixed(1)}</td>
                          <td className="p-3 text-slate-400 font-black text-[10px] tracking-tighter">{r.time}</td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                )}
              </div>
              
              <div className="p-5 bg-slate-50 border-t flex-shrink-0">
                <button 
                  onClick={() => registered.length > 0 ? addToast('학사 서버에 최종 저장되었습니다.', 'success') : addToast('신청 내역이 없습니다.', 'error')}
                  className="w-full bg-[#3344aa] text-white py-4 rounded-2xl font-black text-sm hover:bg-[#2a3a9a] transition shadow-xl active:scale-[0.98] uppercase tracking-wider"
                >
                  Confirm Registration
                </button>
              </div>
            </div>

            {/* 비주얼 시간표 */}
            <div className="h-[60%] bg-white rounded-[32px] shadow-sm border border-slate-200 p-5 overflow-hidden flex flex-col">
              <h4 className="text-[10px] font-black text-slate-400 uppercase tracking-widest mb-4 flex justify-between items-center">
                <span className="flex items-center gap-2"><Calendar className="text-[#3344aa]" size={14} /> Live Timetable Preview</span>
                <span className="text-blue-600 font-black tracking-widest">2025 SEM 1</span>
              </h4>
              <div className="flex-grow border border-slate-200 rounded-3xl overflow-hidden flex flex-col relative bg-slate-50/50 shadow-inner">
                <div className="grid grid-cols-6 border-b border-slate-200 bg-slate-100/50 font-black text-[9px] text-slate-500 text-center items-center h-10">
                  <div className="border-r border-slate-200 bg-slate-200/50 h-full flex items-center justify-center text-[8px] uppercase">Slot</div>
                  <div>MON</div><div>TUE</div><div>WED</div><div>THU</div><div>FRI</div>
                </div>
                <div className="flex-grow relative overflow-y-auto custom-scroll" id="timetable-wrapper">
                  <div className="absolute inset-0 grid grid-rows-9 pointer-events-none">
                    {[1,2,3,4,5,6,7,8,9].map(n => (
                      <div key={n} className="timetable-cell flex items-center justify-center text-[9px] text-slate-300 border-r bg-white/50 font-black">{n}</div>
                    ))}
                  </div>
                  <div className="absolute inset-0 grid grid-cols-6 pointer-events-none">
                    {[1,2,3,4,5].map(n => <div key={n} className="border-r border-slate-100"></div>)}
                    <div></div>
                  </div>
                  
                  {/* 수업 블록 렌더링 */}
                  <div className="absolute inset-0 left-[16.6%] pointer-events-none">
                    {registered.map((r) => {
                      const dayMap = { '월': 0, '화': 1, '수': 2, '목': 3, '금': 4 };
                      const day = dayMap[r.time[0]];
                      const p = r.time.substring(1).split(',').map(Number);
                      const start = p[0];
                      const len = p.length;
                      return (
                        <div 
                          key={r.id}
                          className="course-block animate-in"
                          style={{
                            backgroundColor: r.color,
                            left: `${(day * 20)}%`,
                            top: `${(start - 1) * 42}px`,
                            width: '19.5%',
                            height: `${(len * 42) - 1.5}px`
                          }}
                        >
                          <div className="leading-tight px-1 overflow-hidden">
                            <div className="truncate w-full">{r.title}</div>
                            <div className="opacity-80 text-[7px] font-medium mt-0.5 uppercase tracking-tighter">{r.room.substring(0,6)}</div>
                          </div>
                        </div>
                      );
                    })}
                  </div>
                </div>
              </div>
            </div>
          </div>
        </div>
      </main>

      {/* 토스트 알림 */}
      <div className="fixed bottom-8 left-1/2 -translate-x-1/2 z-[100] flex flex-col gap-2 w-full max-w-xs pointer-events-none">
        {toasts.map((t) => (
          <div key={t.id} className={`${t.type === 'error' ? 'bg-red-600' : t.type === 'info' ? 'bg-blue-600' : 'bg-slate-900'} text-white px-6 py-4 rounded-[24px] shadow-2xl flex items-center gap-3 animate-in border border-white/10 pointer-events-auto`}>
            {t.type === 'error' ? <AlertCircle size={20} /> : t.type === 'info' ? <Info size={20} /> : <CheckCircle size={20} />}
            <span className="text-sm font-black tracking-tight">{t.msg}</span>
          </div>
        ))}
      </div>

      {/* 로딩 모달 */}
      {isLoading && (
        <div className="fixed inset-0 bg-slate-900/80 backdrop-blur-md z-[200] flex items-center justify-center">
          <div className="bg-white p-12 rounded-[48px] shadow-2xl flex flex-col items-center animate-in">
            <div className="w-20 h-20 border-8 border-slate-100 border-t-[#3344aa] rounded-full animate-spin mb-8"></div>
            <p className="font-black text-[#3344aa] text-3xl uppercase tracking-tighter">Processing...</p>
            <p className="text-slate-400 mt-2 font-bold italic">Registering Hyobin's Selection to Registrar...</p>
          </div>
        </div>
      )}
    </div>
  );
}

function PlusCircle({ size, className }) {
  return <svg width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" className={className}><circle cx="12" cy="12" r="10"></circle><line x1="12" y1="8" x2="12" y2="16"></line><line x1="8" y1="12" x2="16" y2="12"></line></svg>;
}
